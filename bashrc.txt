# Функция для запуска RAG MAKE
rag_make() {
    # Цвета для вывода
    local RED='\033[0;31m'
    local GREEN='\033[0;32m'
    local YELLOW='\033[1;33m'
    local BLUE='\033[0;34m'
    local NC='\033[0m' # No Color
    
    # Сохраняем текущую директорию
    local current_dir=$(pwd)
    local rag_dir="$HOME/git/rag"
    
    # Переходим в директорию проекта
    cd "$rag_dir" || {
        echo -e "${RED}❌ Ошибка: папка $rag_dir не найдена${NC}"
        return 1
    }
    
    # Проверка существования виртуального окружения
    if [ ! -d "venv" ]; then
        echo -e "${RED}❌ Ошибка: виртуальное окружение venv не найдено${NC}"
        cd "$current_dir"
        return 1
    fi
    
    # Проверка существования скрипта
    if [ ! -f "make.py" ]; then
        echo -e "${RED}❌ Ошибка: скрипт make.py не найден${NC}"
        cd "$current_dir"
        return 1
    fi
    
    echo -e "${BLUE}🚀 Запуск RAG MAKE...${NC}"
    echo -e "${BLUE}📁 Директория: $rag_dir${NC}"
    echo ""
    
    # Активация виртуального окружения
    source venv/bin/activate
    
    # Запуск скрипта
    python make.py
    
    # Проверяем код возврата скрипта
    local exit_code=$?
    
    if [ $exit_code -eq 0 ]; then
        # Скрипт завершился успешно
        echo ""
        echo "======================================================================"
        echo -e "${YELLOW}🔄 Синхронизировать базу данных RAG с Яндекс.Диском? (Д/Н):${NC} \c"
        read sync_response
        
        case "$sync_response" in
            [YyДд])
                echo ""
                echo -e "${BLUE}🔄 Запуск синхронизации с Яндекс.Диском...${NC}"
                
                # Проверка, запущен ли yandex-disk
                if ! pgrep -x "yandex-disk" > /dev/null; then
                    echo -e "${YELLOW}⚠️  Яндекс.Диск не запущен, запускаю...${NC}"
                    yandex-disk start
                    sleep 2
                fi
                
                yandex-disk sync
                
                if [ $? -eq 0 ]; then
                    echo -e "${GREEN}✓ Синхронизация завершена успешно${NC}"
                    
                    # Показываем статус
                    echo ""
                    echo -e "${BLUE}📊 Статус синхронизации:${NC}"
                    yandex-disk status
                else
                    echo -e "${YELLOW}⚠️  Предупреждение: возникли проблемы при синхронизации${NC}"
                fi
                ;;
            *)
                echo -e "${RED}❌ Синхронизация отменена${NC}"
                ;;
        esac
    else
        echo ""
        echo -e "${RED}⚠️  Скрипт завершился с ошибкой (код: $exit_code)${NC}"
    fi
    
    # Деактивация виртуального окружения
    deactivate
    
    # Возвращаемся в исходную директорию
    cd "$current_dir"
    
    echo ""
    echo -e "${GREEN}👋 Завершено${NC}"
}
# Функция для запуска поиска RAG
rag_search() {
    local current_dir=$(pwd)
    cd $HOME/git/rag || return 1

    if [ ! -f "search.py" ]; then
        echo "❌ Ошибка: скрипт search.py не найден"
        cd "$current_dir"
        return 1
    fi

    source venv/bin/activate
    python search.py
    deactivate
    cd "$current_dir"
}
